if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ReconstructionFilters LANGUAGES CXX)
endif()

add_executable(ReconstructionFilters main.cpp)
target_compile_features(ReconstructionFilters PRIVATE cxx_std_17)
target_include_directories(ReconstructionFilters PRIVATE ../../include)
target_link_libraries(ReconstructionFilters PRIVATE Threads::Threads)
