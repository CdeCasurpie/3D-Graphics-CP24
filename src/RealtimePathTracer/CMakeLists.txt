cmake_minimum_required(VERSION 3.24)
project(RealtimePathTracer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(embree 4 REQUIRED CONFIG)
option(ENABLE_OPTIX "Enable NVIDIA OptiX backend (requires SDK, toolkit, NVIDIA driver/GPU)" OFF)
option(BUILD_REALTIME_VIEWER "Build GLFW/ImGui interactive viewer" ON)
set(OPTIX_ROOT "" CACHE PATH "Extracted NVIDIA OptiX SDK root")
set(OPTIX_COMPUTE_ARCH "75" CACHE STRING "PTX compute capability, digits only (e.g. 75 or 86)")

add_library(RealtimeCore STATIC EmbreeBackend.cpp IO.cpp tests.cpp)
target_include_directories(RealtimeCore PUBLIC ../../include)
target_link_libraries(RealtimeCore PUBLIC embree Threads::Threads)
if(ENABLE_OPTIX)
    find_package(CUDAToolkit REQUIRED)
    find_path(OPTIX_INCLUDE_DIR optix.h HINTS "${OPTIX_ROOT}" "$ENV{OPTIX_ROOT}" PATH_SUFFIXES include REQUIRED)
    if(NOT CUDAToolkit_NVCC_EXECUTABLE)
        message(FATAL_ERROR "OptiX device programs need nvcc from a complete CUDA Toolkit")
    endif()
    if(NOT OPTIX_COMPUTE_ARCH MATCHES "^[0-9]+$")
        message(FATAL_ERROR "OPTIX_COMPUTE_ARCH must be digits, for example 75 or 86")
    endif()
    set(PTX_FILE "${CMAKE_CURRENT_BINARY_DIR}/PathTracerKernel.ptx")
    set(HOST_COMPILER_OPTION)
    if(CMAKE_CUDA_HOST_COMPILER)
        list(APPEND HOST_COMPILER_OPTION -ccbin "${CMAKE_CUDA_HOST_COMPILER}")
    endif()
    add_custom_command(OUTPUT "${PTX_FILE}"
        COMMAND "${CUDAToolkit_NVCC_EXECUTABLE}" --ptx --std=c++17 --fmad=false -O3 -lineinfo
            "--gpu-architecture=compute_${OPTIX_COMPUTE_ARCH}" ${HOST_COMPILER_OPTION}
            "-I${OPTIX_INCLUDE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/PathTracerKernel.cu" -o "${PTX_FILE}"
        DEPENDS PathTracerKernel.cu LaunchParams.h Core.h ../SimplePathTracer/TracerCore.h ../ShootingRaysCUDA/CudaMath.h
        VERBATIM)
    add_custom_target(RealtimePTX DEPENDS "${PTX_FILE}")
    target_sources(RealtimeCore PRIVATE OptixBackend.cpp)
    target_include_directories(RealtimeCore PRIVATE "${OPTIX_INCLUDE_DIR}")
    target_compile_definitions(RealtimeCore PUBLIC REALTIME_OPTIX=1 REALTIME_PTX_PATH="${PTX_FILE}")
    target_link_libraries(RealtimeCore PUBLIC CUDA::cuda_driver ${CMAKE_DL_LIBS})
    add_dependencies(RealtimeCore RealtimePTX)
endif()
add_executable(PathTracer main.cpp)
target_link_libraries(PathTracer PRIVATE RealtimeCore)
if(BUILD_REALTIME_VIEWER)
    find_package(glfw3 REQUIRED)
    find_package(OpenGL REQUIRED)
    find_path(GLM_INCLUDE_DIR glm/glm.hpp REQUIRED)
    add_executable(PathTracerViewer main.cpp Viewer.cpp ../../glad/src/gl.c
        ../imgui/imgui.cpp ../imgui/imgui_draw.cpp ../imgui/imgui_tables.cpp
        ../imgui/imgui_widgets.cpp ../imgui/imgui_impl_glfw.cpp ../imgui/imgui_impl_opengl3.cpp)
    target_compile_definitions(PathTracerViewer PRIVATE REALTIME_VIEWER=1)
    target_include_directories(PathTracerViewer PRIVATE ../../glad/include ../../include/imgui "${GLM_INCLUDE_DIR}")
    target_link_libraries(PathTracerViewer PRIVATE RealtimeCore glfw OpenGL::GL)
endif()
enable_testing()
add_test(NAME EmbreePathTracerTests COMMAND PathTracer --self-test --backend embree)
