# Standalone entrypoint avoids the root project's GLFW/OpenGL requirements.
cmake_minimum_required(VERSION 3.10)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SimplePathTracer LANGUAGES CXX)
endif()
find_package(Threads REQUIRED)
option(ENABLE_CUDA_PATH_TRACER "Build Task 15 CUDA backend" OFF)

set(PATH_TRACER_HOST_SOURCES main.cpp CpuRenderer.cpp tests.cpp)
add_executable(SimplePathTracerCPU ${PATH_TRACER_HOST_SOURCES})
target_compile_features(SimplePathTracerCPU PRIVATE cxx_std_17)
target_include_directories(SimplePathTracerCPU PRIVATE ../../include)
target_link_libraries(SimplePathTracerCPU PRIVATE Threads::Threads)

if(ENABLE_CUDA_PATH_TRACER)
    if(CMAKE_VERSION VERSION_LESS 3.24)
        message(FATAL_ERROR "Task 15 CUDA requires CMake >= 3.24 for native GPU detection")
    endif()
    include(CheckLanguage)
    check_language(CUDA)
    if(NOT CMAKE_CUDA_COMPILER)
        message(FATAL_ERROR "Task 15 CUDA needs nvcc; load the lab CUDA toolkit or disable ENABLE_CUDA_PATH_TRACER")
    endif()
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES native)
    endif()
    cmake_policy(SET CMP0104 NEW)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_executable(SimplePathTracerCUDA ${PATH_TRACER_HOST_SOURCES} CudaRenderer.cu)
    target_compile_features(SimplePathTracerCUDA PRIVATE cxx_std_17)
    set_target_properties(SimplePathTracerCUDA PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_definitions(SimplePathTracerCUDA PRIVATE PATH_TRACER_CUDA=1)
    target_include_directories(SimplePathTracerCUDA PRIVATE ../../include)
    target_link_libraries(SimplePathTracerCUDA PRIVATE Threads::Threads CUDA::cudart)
    target_compile_options(SimplePathTracerCUDA PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
endif()
